      for (let k = 0; k < 4; k++) {

        c.width = w; c.height = Math.round(img.height * w / img.width);
        c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
        url = c.toDataURL('image/png');
        if (url.length < 45000) break;
        w = Math.round(w * 0.75);
      }
      if (url.length >= 45000) return toast('ပုံ အရမ်းကြီးနေပါတယ်၊ ပုံသေးသေး ရွေးပါ', 'bad');
      try {
        const r = await call('appSetLogo', TOKEN, url);
        toast(r.msg, r.ok ? 'ok' : 'bad');
        if (r.ok) setLogo(r.logo);
      } catch (e) { toast(e.message, 'bad'); }
    };
    img.src = rd.result;
  };
  rd.readAsDataURL(f);
}

function getTok() { try { return localStorage.getItem(LS); } catch (e) { return TOKEN; } }
function setTok(t) { TOKEN = t; try { t ? localStorage.setItem(LS, t) : localStorage.removeItem(LS); } catch (e) {} }

// server ကို ခေါ်တဲ့ function — login ကုန်သွားရင် Login စာမျက်နှာ ပြန်ပြ
function call(fn, ...args) {
  return new Promise((res, rej) => {
    google.script.run
      .withSuccessHandler(res)
      .withFailureHandler(e => {
        const m = (e && e.message) || String(e);
        if (m.indexOf('AUTH') > -1) { setTok(null); showLogin(); rej(new Error('ပြန်ဝင်ပေးပါ')); }
        else rej(new Error(m));
      })[fn](...args);
  });
}

function cls(s) {
  if (s.exited) return 'p-off';
  if (s.months === 0 || (s.days !== null && s.days < 0)) return 'p-bad';
  if (s.days !== null && s.days <= D.warn) return 'p-warn';
  return 'p-ok';
}
function pillText(s) {
  if (s.exited) return 'ထွက်ပြီး';
  if (s.months === 0) return 'မပေးရသေး';
  if (s.days === null) return '-';
  if (s.days < 0) return 'ကျော် ' + (-s.days) + ' ရက်';
  return s.days + ' ရက်ကျန်';
}
function short(c) { return String(c).split(' (')[0] + (String(c).indexOf('ကလေး') > -1 ? ' (ကလေး)' : ''); }
function card(s) {
  const i = D.students.findIndex(x => x.name === s.name);

    <span class="pill ${cls(s)}">${pillText(s)}</span></div>`;
}
